A small, complete example

Code is part of the argument. IBM Plex Mono gives it a distinct voice, with readable 0O, 1Il, {}[](), and => != <= at a practical size. The font files ship with the theme, so they work without a font service.

catalog.py
from dataclasses import dataclass
from pathlib import Path

@dataclass(frozen=True)
class Book:
    title: str
    author: str
    year: int

def find_books(books: list[Book], query: str) -> list[Book]:
    """Find a title or author, ignoring case."""
    needle = query.casefold()
    return [
        book for book in books
        if needle in f"{book.title} {book.author}".casefold()
    ]

books = [Book("A Room with a View", "E. M. Forster", 1908)]
print(find_books(books, "forster"))  # One good place to begin

def validate(book: Book) -> None:
    if not book.title.strip():
        raise ValueError("A book needs a title")
    if book.year < 1:
        raise ValueError("The year must be positive")
